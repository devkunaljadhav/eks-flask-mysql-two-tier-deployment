FLASK - MYSQL - EKS DevOps Project
==================================


Launch EC2 amazon Linux 2023 , t2.micro and attach admin role



Install Kubectl on machine
===========================


vi kubectl.sh

sudo yum update -y
curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
chmod +x kubectl && sudo mv kubectl /usr/local/bin/
kubectl version --client

curl --silent --location "https://github.com/eksctl-io/eksctl/releases/latest/download/eksctl_Linux_amd64.tar.gz" | tar xz -C /tmp
sudo mv /tmp/eksctl /usr/local/bin
eksctl version
aws --version

sh kubectl.sh

Now Create EKS Cluster
===================
eksctl create cluster --name kunal-cluster --region ap-south-1 \
  --node-type t3.small --nodes-min 2 --nodes-max 3
aws eks update-kubeconfig --name kunal-cluster --region ap-south-1
kubectl get nodes
=====================

install GIT
=====================
sudo yum install git -y
git clone https://github.com/devkunaljadhav/eks-flask-mysql-project.git
cd eks-flask-mysql-project/eks-manifests/

kubectl apply -f mysql-secrets.yml -f mysql-configmap.yml -f mysql-deployment.yml -f mysql-svc.yml
kubectl apply -f two-tier-app-deployment.yml -f two-tier-app-svc.yml
kubectl get all

kubectl get deploy

=======================
Troubleshooting

kubectl get pods
kubectl logs <pod-name>
kubectl describe pod <pod-name>
kubectl rollout restart deployment two-tier-app

sudo systemctl start docker

cd ~/eks-flask-mysql-project

================
Cleanup (must!)
kubectl delete -f two-tier-app-svc.yml -f two-tier-app-deployment.yml
kubectl delete -f mysql-svc.yml -f mysql-deployment.yml
eksctl delete cluster --name kunal-cluster --region ap-south-1
